☁️ Google Sheets Sync

Store your data in a Google Sheet so it's accessible from any device. Follow these steps once:

1

Go to sheets.google.com and create a new spreadsheet. Name it anything, e.g. Leisurelog.

2

Click Extensions → Apps Script. Delete the placeholder code, paste the script below, then click Save (💾).

3

In the script, replace your-secret-token-here on line 1 with any strong passphrase you choose (e.g. dario-lemon-42!). Only requests that include this token will be accepted — anyone who finds the URL without the token is blocked.

4

Click Deploy → New deployment. Choose type Web app. Set Execute as → Me, Who has access → Anyone. Click Deploy and authorise when prompted.

5

Copy the Web app URL and paste it below along with the same secret token you chose in step 3. Hit Connect.

const SECRET = 'your-secret-token-here'; // ← change this! const SHEET_NAME = 'Entries'; const COLS = ['id','title','type','status','creator','rating', 'progress','tags','notes','dateAdded','dateStarted','dateFinished']; function doGet(e) { var params = e.parameter || {}; if (params.token !== SECRET) return forbidden(); if (params.action === 'load') return respond(loadEntries()); return respond({ error: 'Unknown action' }); } function doPost(e) { var body = JSON.parse(e.postData.contents); if (body.token !== SECRET) return forbidden(); if (body.action === 'save') { saveEntries(body.entries || []); return respond({ ok: true, saved: body.entries.length }); } return respond({ error: 'Unknown action' }); } function forbidden() { return ContentService .createTextOutput(JSON.stringify({ error: 'Unauthorized' })) .setMimeType(ContentService.MimeType.JSON); } function loadEntries() { var sheet = getSheet(); var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return { entries: [] }; var entries = rows.slice(1).map(function(row) { var entry = {}; COLS.forEach(function(c, i) { entry[c] = row[i] === '' ? null : row[i]; }); entry.tags = entry.tags ? String(entry.tags).split(',') : []; entry.rating = entry.rating !== null ? Number(entry.rating) : null; return entry; }); return { entries: entries }; } function saveEntries(entries) { var sheet = getSheet(); sheet.clearContents(); sheet.appendRow(COLS); entries.forEach(function(entry) { sheet.appendRow(COLS.map(function(c) { if (c === 'tags') return (entry[c] || []).join(','); return entry[c] == null ? '' : entry[c]; })); }); } function getSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(COLS); } return sheet; } function respond(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
